
CREATE DATABASE shunda DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

CREATE USER `shunda`@`localhost` IDENTIFIED BY 'shunda';
GRANT SELECT ON `mysql`.* TO `shunda` @`localhost`;
GRANT ALL ON `shunda`.* TO `shunda` @`localhost`;

CREATE USER `shunda`@`%` IDENTIFIED BY 'shunda';
GRANT SELECT ON `mysql`.* TO `shunda` @`%`;
GRANT ALL ON `shunda`.* TO `shunda` @`%`;
